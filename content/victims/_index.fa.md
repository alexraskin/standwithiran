lastmod = "2026-11-17"

هر فردی که در زیر آمده است توسط سازمانی مستقل مستند شده و به پرونده‌ی او پیوند داده شده است. این‌ها گزیده‌ی کوچکی از هزاران نفرند. پرونده‌های کامل نزد سازمان‌هایی است که در پایین این صفحه آمده‌اند.
